Next.js FastAPI SQLite

풀스택 미니 쇼핑몰 장바구니

Next.js(프론트) + FastAPI(백엔드) + SQLite(DB)로 처음 만들어본 풀스택 웹앱입니다. 영수증 스타일 UI로 상품을 담고, 수량을 바꾸고, 삭제하는 것까지 전부 실제 DB에 반영됩니다.

종합 실습

미니 쇼핑몰 장바구니

상품 목록 표시, 장바구니 담기(중복 시 수량+1), 수량 +/- 변경, 개별/전체 삭제, 총 금액 자동 계산까지 되는 실제 동작하는 쇼핑몰 카트

  • API : GET /products, GET /cart, POST /cart, PATCH /cart/{id}, DELETE /cart/{id}, DELETE /cart
  • 실행 방법 : backend는 uvicorn main:app --reload (127.0.0.1:8000), frontend는 npm run dev (localhost:3000)

def _fetch_cart_row(conn, cart_id: int):
    return conn.execute(
        """
        SELECT cart.id AS id,
               cart.product_id AS product_id,
               products.name AS name,
               products.price AS price,
               cart.quantity AS quantity
        FROM cart
        JOIN products ON products.id = cart.product_id
        WHERE cart.id = ?
        """,
        (cart_id,),
    ).fetchone()


@app.get("/products", summary="상품목록가져오기")
def get_products() -> list[Product]:
    conn = get_connection()
    rows = conn.execute("SELECT id, name, price FROM products").fetchall()
    conn.close()
    return [Product(**dict(row)) for row in rows]


@app.get("/cart", summary="장바구니가져오기")
def get_cart() -> list[CartItemOut]:
    conn = get_connection()
    rows = conn.execute(
        """
        SELECT cart.id AS id,
               cart.product_id AS product_id,
               products.name AS name,
               products.price AS price,
               cart.quantity AS quantity
        FROM cart
        JOIN products ON products.id = cart.product_id
        """
    ).fetchall()
    conn.close()
    return [CartItemOut(**dict(row)) for row in rows]


@app.post("/cart", summary="장바구니에상품추가")
def add_to_cart(item: CartItemCreate) -> CartItemOut:
    conn = get_connection()
    existing = conn.execute(
        "SELECT id, quantity FROM cart WHERE product_id = ?", (item.product_id,)
    ).fetchone()

    if existing is None:
        cursor = conn.execute(
            "INSERT INTO cart (product_id, quantity) VALUES (?, 1)",
            (item.product_id,),
        )
        cart_id = cursor.lastrowid
    else:
        cart_id = existing["id"]
        conn.execute(
            "UPDATE cart SET quantity = ? WHERE id = ?",
            (existing["quantity"] + 1, cart_id),
        )

    conn.commit()
    row = _fetch_cart_row(conn, cart_id)
    conn.close()
    return CartItemOut(**dict(row))


@app.patch("/cart/{id}", summary="장바구니수량변경")
def update_cart_quantity(id: int, item: CartItemUpdate) -> CartItemOut:
    quantity = max(1, item.quantity)
    conn = get_connection()
    conn.execute("UPDATE cart SET quantity = ? WHERE id = ?", (quantity, id))
    conn.commit()
    row = _fetch_cart_row(conn, id)
    conn.close()
    return CartItemOut(**dict(row))


@app.delete("/cart/{id}", summary="장바구니에서상품삭제")
def delete_from_cart(id: int):
    conn = get_connection()
    conn.execute("DELETE FROM cart WHERE id = ?", (id,))
    conn.commit()
    conn.close()
    return {"deleted": id}


@app.delete("/cart", summary="장바구니비우기")
def clear_cart():
    conn = get_connection()
    conn.execute("DELETE FROM cart")
    conn.commit()
    conn.close()
    return {"cleared": True}

import sqlite3
from pathlib import Path

DB_PATH = Path(__file__).parent / "cart.db"

INITIAL_PRODUCTS = [
    ("노트북", 1200000),
    ("키보드", 80000),
    ("마우스", 40000),
    ("헤드셋", 100000),
]


def get_connection():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn


def init_db():
    conn = get_connection()
    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS products (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            price INTEGER NOT NULL
        )
        """
    )
    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS cart (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            product_id INTEGER NOT NULL,
            quantity INTEGER NOT NULL,
            FOREIGN KEY (product_id) REFERENCES products (id)
        )
        """
    )

    count = conn.execute("SELECT COUNT(*) FROM products").fetchone()[0]
    if count == 0:
        conn.executemany(
            "INSERT INTO products (name, price) VALUES (?, ?)", INITIAL_PRODUCTS
        )

    conn.commit()
    conn.close()

frontend/app/page.tsx 코드 보기 
"use client";

import { useEffect, useState } from "react";

const API_BASE = "http://127.0.0.1:8000";

type Product = {
  id: number;
  name: string;
  price: number;
};

type CartItem = {
  id: number;
  product_id: number;
  name: string;
  price: number;
  quantity: number;
};

export default function Home() {
  const [products, setProducts] = useState<Product[]>([]);
  const [cart, setCart] = useState<CartItem[]>([]);

  async function fetchProducts() {
    const res = await fetch(`${API_BASE}/products`, { cache: "no-store" });
    setProducts(await res.json());
  }

  async function fetchCart() {
    const res = await fetch(`${API_BASE}/cart`, { cache: "no-store" });
    setCart(await res.json());
  }

  useEffect(() => {
    fetchProducts();
    fetchCart();
  }, []);

  async function handleAdd(productId: number) {
    await fetch(`${API_BASE}/cart`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ product_id: productId }),
    });
    fetchCart();
  }

  async function handleRemove(itemId: number) {
    await fetch(`${API_BASE}/cart/${itemId}`, { method: "DELETE" });
    fetchCart();
  }

  async function handleChangeQuantity(itemId: number, quantity: number) {
    await fetch(`${API_BASE}/cart/${itemId}`, {
      method: "PATCH",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ quantity }),
    });
    fetchCart();
  }

  async function handleClearCart() {
    await fetch(`${API_BASE}/cart`, { method: "DELETE" });
    fetchCart();
  }

  const total = cart.reduce((sum, item) => sum + item.price * item.quantity, 0);

  return (
    <div className="receipt">
      <div className="store-title">미니쇼핑몰</div>
      <div className="subtitle">MINI SHOPPING MALL RECEIPT</div>

      <hr className="divider" />

      <div className="section-title">상품 목록</div>
      {products.map((product) => (
        <div className="line" key={product.id}>
          <span className="label">{product.name}</span>
          <span className="fill" />
          <span className="value">{product.price.toLocaleString()}원</span>
          <button className="receipt-btn" onClick={() => handleAdd(product.id)}>
            [담기]
          </button>
        </div>
      ))}

      <hr className="divider" />

      <div className="section-title">장바구니</div>
      <button className="receipt-btn clear-btn" onClick={handleClearCart}>
        [장바구니 비우기]
      </button>

      {cart.length === 0 ? (
        <p className="empty-cart">장바구니가 비어 있습니다.</p>
      ) : (
        cart.map((item) => (
          <div className="line" key={item.id}>
            <span className="label">{item.name}</span>
            <span className="fill" />
            <span className="value">{item.price.toLocaleString()}원</span>
            <div className="qty-controls">
              <button
                className="receipt-btn"
                onClick={() =>
                  handleChangeQuantity(item.id, item.quantity - 1)
                }
              >
                -
              </button>
              <span>{item.quantity}</span>
              <button
                className="receipt-btn"
                onClick={() =>
                  handleChangeQuantity(item.id, item.quantity + 1)
                }
              >
                +
              </button>
            </div>
            <button className="receipt-btn" onClick={() => handleRemove(item.id)}>
              [삭제]
            </button>
          </div>
        ))
      )}

      <div className="total-line">
        <span>총 금액</span>
        <span>{total.toLocaleString()}원</span>
      </div>
    </div>
  );
}

from pydantic import BaseModel

app = FastAPI()

memos = []

    content: str


@app.get("/")
def read_root():
    return "안녕하세요"


@app.post("/memos")
def create_memo(memo: Memo):
    memos.append(memo)
    return memo


@app.get("/memos")
def list_memos():
    return memos
