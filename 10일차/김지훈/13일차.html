LED 3색 순차 점등

빨강·노랑·파랑 LED 3개를 순서대로 켰다 끄는 기본 회로

코드 보기 
// C++ code
//

int Red=7;
int Yellow=6;
int Blue=5;

void setup()
{
  pinMode(Red, OUTPUT);
  pinMode(Yellow, OUTPUT);
  pinMode(Blue, OUTPUT);
}

void loop()
{
  digitalWrite(Red, LOW);
  digitalWrite(Yellow, LOW);
  digitalWrite(Blue, LOW);
  delay(1000);

  digitalWrite(Red, HIGH);
  delay(1000);
  digitalWrite(Red, LOW);
  digitalWrite(Yellow, HIGH);
  delay(1000);
  pinMode(Yellow, LOW);
  digitalWrite(Blue, HIGH);
  delay(1000);
  digitalWrite(Blue, LOW);
}

PIR 동작 감지 센서

PIR 센서로 움직임을 감지하면 LED를 켜고 시리얼 모니터에 "Welcome!" 출력

코드 보기 
// C++ code

int ledPin = 8;
int inputPin = 7;
int val = 0;

void setup()
{
  pinMode(ledPin, OUTPUT);
  pinMode(inputPin, INPUT);
  Serial.begin(9600);
}

void loop()
{
  val = digitalRead(inputPin);

  if(val == HIGH){
    digitalWrite(ledPin, HIGH);
    Serial.println("Welcome!");
  }
  else{
    digitalWrite(ledPin, LOW);
    Serial.println("Nothing");
  }
  delay(1000);
}

초음파 거리 센서 (HC-SR04)

초음파 센서로 물체까지의 거리를 mm 단위로 계산해 시리얼 모니터에 출력

코드 보기 
// C++ code

int echoPin = 7;
int trigPin = 8;

void setup()
{
  Serial.begin(9600);
  pinMode(echoPin, INPUT);
  pinMode(trigPin, OUTPUT);
}

void loop()
{
  long duration, distance;
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);
  duration = pulseIn(echoPin, HIGH);
  distance = ((float)(340*duration) /1000) / 2;

  Serial.print("Distance:");
  Serial.print(distance);
  Serial.print("mm/n");
  delay(1000);
}

종합 과제 — 물체 감지 알람

초음파 센서 + 부저 + LED를 합쳐서, 300~3000mm 사이에 물체가 감지되면 "물체감지!"를 알리는 알람 회로

코드 보기 
// C++ code
//
int Buzzer = 4;
int Red = 5;
int echoPin = 7;
int trigPin = 8;

long duration;
float distance;

void setup() {
  Serial.begin(9600);
  pinMode(echoPin, INPUT);
  pinMode(trigPin, OUTPUT);
  pinMode(Buzzer, OUTPUT);
  pinMode(Red, OUTPUT);
}

void loop() {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  duration = pulseIn(echoPin, HIGH);
  distance = ((float)(340 * duration) / 1000) / 2;

  if (distance > 300&&distance < 3000) {
    digitalWrite(Red, HIGH);
    tone(Buzzer, 523);
    Serial.println("물체감지!");
  } else {
    digitalWrite(Red, LOW);
    noTone(Buzzer);
    Serial.println("Nothing");
  }

  delay(300);
}
