삼항 연산자 활용

prompt 입력값으로 적정 체중 여부 판정하기

새 탭에서 열기 →
index.html 코드 보기 
!DOCTYPE html>
<head>
    <meta http-equiv="content-type" content="tete/html;" charset="utf-8">
    <title>변수 선언 및 출력</title>
    <script type="text/javascript">
        var user_height = prompt("당신의 신장은?","0");
        var user_weight = prompt("당신의 체중은?","0");

        var normal_weight = (user_height-100)*0.9;
        var result = (user_weight >= normal_weight) ? "적정몸무게 이상" : "적정몸무게 미달";

        document.write(result);
    </script>
</head>
