집합 연산 — 교집합 · 합집합 · 차집합

for문 + in 연산자로 직접 구현한 방식과 set 연산자를 이용한 방식을 각각 비교

노트북 내용 보기 
# 1. 교집합 (for문 + in)
A=[23, 15, 2, 14, 14, 16, 20, 14]
B=[2, 48, 15, 14, 26, 32, 47, 14]
C=[]
for item in A:
    if item in B:
        C.append(item)
print(C)
# 실행 결과
[15, 2, 14, 14, 14]

# 2. 교집합 (set.intersection)
A=[23, 15, 2, 14, 14, 16, 20, 14]
B=[2, 48, 15, 14, 26, 32, 47, 14]
C=list(set(A).intersection(B))
print(C)
# 실행 결과
[2, 14, 15]

# 3. 합집합 (리스트 + 연산자, 중복 포함)
A=[23, 15, 2, 14, 14, 16, 20, 14]
B=[2, 48, 15, 14, 26, 32, 47, 14]
U=[]
U=A+B
print(U)
# 실행 결과
[23, 15, 2, 14, 14, 16, 20, 14, 2, 48, 15, 14, 26, 32, 47, 14]

# 4. 합집합 (set.union, 중복 제거)
A=[23, 15, 2, 14, 14, 16, 20, 14]
B=[2, 48, 15, 14, 26, 32, 47, 14]
U=[]
U=list(set(A).union(B))
print(U)
# 실행 결과
[2, 14, 15, 16, 20, 23, 26, 32, 47, 48]

# 5. 차집합 (for문 + not in)
A=[23, 15, 2, 14, 14, 16, 20, 14]
B=[2, 48, 15, 14, 26, 32, 47, 14]
D=[]
for item in A:
    if item not in B:
        D.append(item)
print(D) #in 연산을 이용한 for문으로 구현한 차집합
# 실행 결과
[23, 16, 20]

# 6. 차집합 (set.difference)
A=[23, 15, 2, 14, 14, 16, 20, 14]
B=[2, 48, 15, 14, 26, 32, 47, 14]
D=list(set(A).difference(B))
print(D) #set 연산을 이용한 차집합
# 실행 결과
[16, 20, 23]
