<script>
const fileInput = document.getElementById('fileInput');
const uploadArea = document.getElementById('uploadArea');
const resultArea = document.getElementById('resultArea');
const preview = document.getElementById('preview');
const overlay = document.getElementById('overlay');
const detectBtn = document.getElementById('detectBtn');
const resultText = document.getElementById('resultText');
const resetBtn = document.getElementById('resetBtn');
const ctx = overlay.getContext('2d');

let model = null;

const LABEL_MAP = {
  person: { ko: '사람', color: '#c9a227' },
  dog: { ko: '강아지', color: '#ff9800' }
};

const fileNameDisplay = document.getElementById('fileNameDisplay');

fileInput.addEventListener('change', () => {
  const file = fileInput.files[0];
  if (!file) return;

  fileNameDisplay.textContent = `📄 파일 선택 - ${file.name}`;

  const reader = new FileReader();
  reader.onload = e => {
    preview.src = e.target.result;
  };
  reader.readAsDataURL(file);
});

preview.addEventListener('load', () => {
  uploadArea.style.display = 'none';
  resultArea.style.display = 'flex';
  resultText.textContent = '';
  const w = preview.clientWidth;
  const h = preview.clientHeight;
  overlay.width = w;
  overlay.height = h;
  overlay.style.width = w + 'px';
  overlay.style.height = h + 'px';
  ctx.clearRect(0, 0, w, h);
});

detectBtn.addEventListener('click', async () => {
  detectBtn.disabled = true;
  detectBtn.textContent = model ? '분석 중...' : '모델 불러오는 중...';
  resultText.textContent = '';

  try {
    if (!model) {
      model = await cocoSsd.load();
    }

    detectBtn.textContent = '분석 중...';
    const predictions = await model.detect(preview);

    const w = preview.clientWidth;
    const h = preview.clientHeight;
    overlay.width = w;
    overlay.height = h;
    ctx.clearRect(0, 0, w, h);

    const found = predictions.filter(p => LABEL_MAP[p.class]);

    found.forEach(p => {
      const info = LABEL_MAP[p.class];
      const [x, y, bw, bh] = p.bbox;
      const percent = Math.round(p.score * 100);

      ctx.strokeStyle = info.color;
      ctx.lineWidth = 3;
      ctx.strokeRect(x, y, bw, bh);

      const label = `${info.ko} ${percent}%`;
      ctx.font = '15px sans-serif';
      const textWidth = ctx.measureText(label).width;

      ctx.fillStyle = info.color;
      ctx.fillRect(x, Math.max(y - 22, 0), textWidth + 12, 22);

      ctx.fillStyle = '#fff';
      ctx.fillText(label, x + 6, Math.max(y - 6, 16));
    });

    if (found.length > 0) {
      const names = found.map(p => LABEL_MAP[p.class].ko).join(' ');
      resultText.textContent = `물체를 ${found.length}개 찾았어요: ${names}`;
    } else {
      resultText.textContent = '사람이나 강아지를 찾지 못했어요';
    }
  } catch (err) {
    resultText.textContent = '분석 중 오류가 발생했어요';
    console.error(err);
  } finally {
    detectBtn.disabled = false;
    detectBtn.textContent = '물체찾기';
  }
});

resetBtn.addEventListener('click', () => {
  fileInput.value = '';
  preview.src = '';
  ctx.clearRect(0, 0, overlay.width, overlay.height);
  resultText.textContent = '';
  fileNameDisplay.textContent = '';
  resultArea.style.display = 'none';
  uploadArea.style.display = 'block';
  detectBtn.disabled = false;
  detectBtn.textContent = '물체찾기';
});
</script>
</body>
</html>

  <h1>🐍 스네이크 게임</h1>
  <div id="score">점수: 0</div>
  <canvas id="gameCanvas" width="400" height="400"></canvas>
  <div id="hint">방향키(↑ ↓ ← →)로 이동하세요</div>

  <div id="overlay">
    <h2>게임 오버</h2>
    <p id="finalScore">최종 점수: 0</p>
    <button id="restartBtn">다시 시작</button>

<script>
const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');
const scoreEl = document.getElementById('score');
const overlay = document.getElementById('overlay');
const finalScoreEl = document.getElementById('finalScore');
const restartBtn = document.getElementById('restartBtn');

const gridSize = 20;
const tileCount = canvas.width / gridSize;

let snake, direction, nextDirection, food, score, gameOver, gameLoopId;

function init() {
  snake = [{ x: 10, y: 10 }];
  direction = { x: 0, y: 0 };
  nextDirection = { x: 0, y: 0 };
  score = 0;
  gameOver = false;
  placeFood();
  scoreEl.textContent = '점수: 0';
  overlay.style.display = 'none';
  if (gameLoopId) clearInterval(gameLoopId);
  gameLoopId = setInterval(tick, 120);
}

function placeFood() {
  do {
    food = {
      x: Math.floor(Math.random() * tileCount),
      y: Math.floor(Math.random() * tileCount)
    };
  } while (snake.some(seg => seg.x === food.x && seg.y === food.y));
}

function tick() {
  if (gameOver) return;

  direction = nextDirection;

  if (direction.x === 0 && direction.y === 0) {
    draw();
    return;
  }

  const head = {
    x: snake[0].x + direction.x,
    y: snake[0].y + direction.y
  };

  if (
    head.x < 0 || head.x >= tileCount ||
    head.y < 0 || head.y >= tileCount ||
    snake.some(seg => seg.x === head.x && seg.y === head.y)
  ) {
    endGame();
    return;
  }

  snake.unshift(head);

  if (head.x === food.x && head.y === food.y) {
    score += 10;
    scoreEl.textContent = '점수: ' + score;
    placeFood();
  } else {
    snake.pop();
  }

  draw();
}

function draw() {
  ctx.fillStyle = '#111';
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  ctx.fillStyle = '#ff5555';
  ctx.fillRect(food.x * gridSize, food.y * gridSize, gridSize - 2, gridSize - 2);

  snake.forEach((seg, i) => {
    ctx.fillStyle = i === 0 ? '#4caf50' : '#8bc34a';
    ctx.fillRect(seg.x * gridSize, seg.y * gridSize, gridSize - 2, gridSize - 2);
  });
}

function endGame() {
  gameOver = true;
  clearInterval(gameLoopId);
  finalScoreEl.textContent = '최종 점수: ' + score;
  overlay.style.display = 'flex';
}

document.addEventListener('keydown', e => {
  switch (e.key) {
    case 'ArrowUp':
      if (direction.y === 0) nextDirection = { x: 0, y: -1 };
      e.preventDefault();
      break;
    case 'ArrowDown':
      if (direction.y === 0) nextDirection = { x: 0, y: 1 };
      e.preventDefault();
      break;
    case 'ArrowLeft':
      if (direction.x === 0) nextDirection = { x: -1, y: 0 };
      e.preventDefault();
      break;
    case 'ArrowRight':
      if (direction.x === 0) nextDirection = { x: 1, y: 0 };
      e.preventDefault();
      break;
  }
});

restartBtn.addEventListener('click', init);

init();
</script>
</body>
</html>
